-- Inner Join.

-- List out the students name who paid more than 10000 in a single transaction
select student,paid from payments where paid >15000;
select s.name,p.paid
from students s
inner join payments p
on s.id = p.student
where paid >15000

-- Left join 
list out student names and their payment transaction 

select name from students;

-- Left join 
select s.name,p.*
from students s
left join payments p 
on s.id = p.student

-- right join : 
-- map the students names with payment transaction 

select s.name,p.*
from students s
right join payments p 
on s.id = p.student

-- full join

select s.name,p.*
from students s
full join payments p 
on s.id = p.student
